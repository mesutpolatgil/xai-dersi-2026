In [1]: Şekil 24.1 — Bisiklet Kiralama Verisinde LOFO Özellik Önemi (bike.csv) 
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.ensemble import RandomForestRegressor, RandomForestClassifier
from sklearn.preprocessing import OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from sklearn.metrics import mean_absolute_error, accuracy_score

bike_df = pd.read_csv("bike.csv").dropna().reset_index(drop=True)
bike_features = ["season", "holiday", "workday", "weather", "temp", "hum", "windspeed", "cnt_2d_bfr"]

np.random.seed(42)
perm_b = np.random.permutation(len(bike_df))
n_tr_b = int(len(bike_df) * 2 / 3)
bike_train, bike_test = bike_df.iloc[perm_b[:n_tr_b]], bike_df.iloc[perm_b[n_tr_b:]]
y_tr_b, y_te_b = bike_train["cnt"].values, bike_test["cnt"].values

def fit_eval_rf_bike(train_df, test_df, cols):
    cat_c = [c for c in cols if c in ["season", "holiday", "workday", "weather"]]
    num_c = [c for c in cols if c not in cat_c]
    pre = ColumnTransformer([("cat", OneHotEncoder(drop="first"), cat_c), ("num", "passthrough", num_c)])
    pipe = Pipeline([("pre", pre), ("rf", RandomForestRegressor(n_estimators=100, random_state=42))])
    pipe.fit(train_df[cols], y_tr_b)
    return mean_absolute_error(y_te_b, pipe.predict(test_df[cols]))

orig_mae = fit_eval_rf_bike(bike_train, bike_test, bike_features)
print(f"Tam Model Orijinal Test Hatası (e_orig): {orig_mae:.2f} MAE\n")

lofo_rows = []
for col in bike_features:
    rem_cols = [c for c in bike_features if c != col]
    err_without = fit_eval_rf_bike(bike_train, bike_test, rem_cols)
    lofo_rows.append({
        "feature": col,
        "err_without": err_without,
        "lofo_diff": err_without - orig_mae,
        "lofo_ratio": err_without / orig_mae
    })

lofo_bike_df = pd.DataFrame(lofo_rows).sort_values("lofo_diff", ascending=False)
print(lofo_bike_df.to_string(index=False))

In [3]: Şekil 24.3 — Penguen Cinsiyeti Sınıflandırmasında LOFO (penguins.csv) 
penguins_df = pd.read_csv("penguins.csv").dropna().reset_index(drop=True)
all_p_cols = ["species", "bill_length_mm", "bill_depth_mm", "flipper_length_mm", "body_mass_g"]

np.random.seed(42)
perm_p = np.random.permutation(len(penguins_df))
n_tr_p = int(len(penguins_df) * 2 / 3)
peng_train, peng_test = penguins_df.iloc[perm_p[:n_tr_p]], penguins_df.iloc[perm_p[n_tr_p:]]
y_te_p = (peng_test["sex"] == "male").astype(int).values

def train_eval_peng_rf(cols_to_use):
    cat_c = [c for c in cols_to_use if c == "species"]
    num_c = [c for c in cols_to_use if c != "species"]
    transformers = []
    if cat_c: transformers.append(("cat", OneHotEncoder(drop="first"), cat_c))
    if num_c: transformers.append(("num", "passthrough", num_c))
    pipe = Pipeline([("pre", ColumnTransformer(transformers)), ("rf", RandomForestClassifier(n_estimators=100, random_state=42))])
    pipe.fit(peng_train[cols_to_use], (peng_train["sex"] == "male").astype(int))
    return accuracy_score(y_te_p, pipe.predict(peng_test[cols_to_use]))

orig_acc = train_eval_peng_rf(all_p_cols)
lofo_peng_rows = []
for col in all_p_cols:
    rem_cols = [c for c in all_p_cols if c != col]
    acc_without = train_eval_peng_rf(rem_cols)
    # Doğruluk (accuracy) daha büyük olduğunda daha iyi olduğu için fark -1 ile çarpılır
    lofo_val = -1.0 * (acc_without - orig_acc)
    lofo_peng_rows.append({"feature": col, "acc_without": acc_without, "lofo_importance": lofo_val})

lofo_peng_df = pd.DataFrame(lofo_peng_rows).sort_values("lofo_importance", ascending=False)
print(f"Tam Model Orijinal Doğruluk (Accuracy): {orig_acc:.4f}\n")
print(lofo_peng_df.to_string(index=False))

In [4]: Bölüm 24.6 "Yazılım ve Alternatifler" — lofo-importance Python Kütüphanesi ile Kullanım 
# Kitabın 24.6 "Yazılım ve Alternatifler" bölümünde önerilen `lofo-importance` Python paketi (pip install lofo-importance)
# Kaynak: https://github.com/aerdem4/lofo-importance
from lofo import LOFOImportance, Dataset, plot_importance
from sklearn.ensemble import RandomForestRegressor
from sklearn.model_selection import KFold

# 1. Veri kümesini ve hedef değişkeni tanımla
dataset = Dataset(
    df=bike_df,
    target="cnt",
    features=["season", "holiday", "workday", "weather", "temp", "hum", "windspeed", "cnt_2d_bfr"]
)

# 2. Model ve değerlendirme (Cross-Validation veya ayrılmış test kümesi) ile LOFOImportance nesnesini oluştur
rf_model = RandomForestRegressor(n_estimators=100, random_state=42)
lofo_imp = LOFOImportance(
    dataset=dataset,
    model=rf_model,
    cv=KFold(n_splits=4, shuffle=True, random_state=42),
    scoring="neg_mean_absolute_error"
)

# 3. Önem değerlerini DataFrame olarak al ve grafiğini çizdir
importance_df = lofo_imp.get_importance()
print(importance_df[["feature", "importance_mean", "importance_std"]].to_string(index=False))
