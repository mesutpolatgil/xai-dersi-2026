import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.svm import SVR
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import mean_absolute_error, log_loss
from scipy.stats import pearsonr

# 1. Orijinal CSV dosyalarını yükle ve eksik gözlemleri (NA) temizle
bike_df = pd.read_csv("bike.csv").dropna().reset_index(drop=True)
penguins_df = pd.read_csv("penguins.csv").dropna().reset_index(drop=True)

bike_features = ["season", "holiday", "workday", "weather", "temp", "hum", "windspeed", "cnt_2d_bfr"]
penguin_features = ["species", "bill_length_mm", "bill_depth_mm", "flipper_length_mm", "body_mass_g"]

# 2. Kitaptaki gibi verinin 2/3'ünü eğitim, 1/3'ünü PFI hesabı (test) için ayır
np.random.seed(42)
perm_b = np.random.permutation(len(bike_df))
n_tr_b = int(len(bike_df) * 2 / 3)
bike_train, bike_test = bike_df.iloc[perm_b[:n_tr_b]], bike_df.iloc[perm_b[n_tr_b:]]

perm_p = np.random.permutation(len(penguins_df))
n_tr_p = int(len(penguins_df) * 2 / 3)
peng_train, peng_test = penguins_df.iloc[perm_p[:n_tr_p]], penguins_df.iloc[perm_p[n_tr_p:]]

print(f"bike.csv     -> Toplam: {len(bike_df)}, Eğitim: {len(bike_train)}, Test: {len(bike_test)}")
print(f"penguins.csv -> Toplam: {len(penguins_df)}, Eğitim: {len(peng_train)}, Test: {len(peng_test)}")
print("\n--- bike.csv İlk 5 Gözlem ---")
print(bike_df[bike_features + ["cnt"]].head(5).to_string(index=False))

In [2]: Şekil 23.1 — Destek Vektör Makinesi (SVM) ile Bisiklet Tahmininde PFI 
cat_cols = ["season", "holiday", "workday", "weather"]
num_cols = ["temp", "hum", "windspeed", "cnt_2d_bfr"]

preprocessor = ColumnTransformer([
    ("cat", OneHotEncoder(drop="first"), cat_cols),
    ("num", StandardScaler(), num_cols)
])

svm_model = Pipeline([
    ("prep", preprocessor),
    ("svr", SVR(kernel="rbf", C=1200, gamma="scale"))
])
svm_model.fit(bike_train[bike_features], bike_train["cnt"].values)

y_te_b = bike_test["cnt"].values
orig_mae = mean_absolute_error(y_te_b, svm_model.predict(bike_test[bike_features]))
print(f"Orijinal Test Hatası (e_orig): {orig_mae:.2f} MAE\n")

# Her özelliği K=30 kez permütasyona tabi tutarak FI = e_perm / e_orig oranını ve %5-%95 kantillerini hesapla
pfi_records = []
for col in bike_features:
    ratios = []
    for _ in range(30):
        X_perm = bike_test[bike_features].copy()
        X_perm[col] = np.random.permutation(X_perm[col].values)
        perm_mae = mean_absolute_error(y_te_b, svm_model.predict(X_perm))
        ratios.append(perm_mae / orig_mae)
    pfi_records.append({
        "feature": col,
        "q05": np.quantile(ratios, 0.05),
        "importance": np.mean(ratios),
        "q95": np.quantile(ratios, 0.95),
        "perm_mae": np.mean(ratios) * orig_mae
    })

pfi_bike_df = pd.DataFrame(pfi_records).sort_values("importance", ascending=False)
print(pfi_bike_df.to_string(index=False))

In [3]: Şekil 23.2 — Penguen Cinsiyeti Sınıflandırmasında PFI (Log Loss) 
num_p_cols = ["bill_length_mm", "bill_depth_mm", "flipper_length_mm", "body_mass_g"]
all_p_cols = ["species"] + num_p_cols

# Kitaptaki gibi her tür (Adelie, Chinstrap, Gentoo) için ayrı bir Lojistik Regresyon eğiten Kara Kutu Model
class SpeciesBlackBoxLogReg:
    def __init__(self):
        self.models = {}
    def fit(self, df):
        for sp in ["Adelie", "Chinstrap", "Gentoo"]:
            sub = df[df["species"] == sp]
            pipe = Pipeline([("scaler", StandardScaler()), ("clf", LogisticRegression())])
            pipe.fit(sub[num_p_cols], (sub["sex"] == "male").astype(int))
            self.models[sp] = pipe
        return self
    def predict_proba(self, df):
        probs = np.zeros((len(df), 2))
        for sp, pipe in self.models.items():
            mask = (df["species"] == sp).values
            if np.any(mask):
                probs[mask] = pipe.predict_proba(df.loc[mask, num_p_cols])
        return np.clip(probs, 1e-6, 1 - 1e-6)

bb_model = SpeciesBlackBoxLogReg().fit(peng_train)
y_te_p = (peng_test["sex"] == "male").astype(int).values
orig_ll = log_loss(y_te_p, bb_model.predict_proba(peng_test))

pfi_peng = []
for col in all_p_cols:
    ratios = []
    for _ in range(30):
        X_perm = peng_test.copy()
        X_perm[col] = np.random.permutation(X_perm[col].values)
        ratios.append(log_loss(y_te_p, bb_model.predict_proba(X_perm)) / orig_ll)
    pfi_peng.append({
        "feature": col,
        "q05": np.quantile(ratios, 0.05),
        "importance": np.mean(ratios),
        "q95": np.quantile(ratios, 0.95)
    })

pfi_peng_df = pd.DataFrame(pfi_peng).sort_values("importance", ascending=False)
print(f"Orijinal Test Log Loss: {orig_ll:.4f}\n")
print(pfi_peng_df.to_string(index=False))

# 2. Şekil 23.4: Her penguen türünün yalnızca kendi alt kümesi içinde karıştırılması (Grup Bazlı PFI)
group_pfi_rows = []
for sp in ["Adelie", "Chinstrap", "Gentoo"]:
    sub_te = peng_test[peng_test["species"] == sp].copy()
    y_sub = (sub_te["sex"] == "male").astype(int).values
    base_ll = log_loss(y_sub, bb_model.predict_proba(sub_te))
    for col in num_p_cols:
        ratios = []
        for _ in range(30):
            perm_sub = sub_te.copy()
            perm_sub[col] = np.random.permutation(perm_sub[col].values)
            ratios.append(log_loss(y_sub, bb_model.predict_proba(perm_sub)) / base_ll)
        group_pfi_rows.append({"species": sp, "feature": col, "importance": np.mean(ratios)})

group_pfi_pivot = pd.DataFrame(group_pfi_rows).pivot(index="feature", columns="species", values="importance")

# Kitabın 23.7 "Yazılım ve Alternatifler" bölümünde önerilen hazır Python kütüphaneleriyle PFI hesabı:
# 1) scikit-learn: sklearn.inspection.permutation_importance (Fark ölçütü: e_perm - e_orig)
from sklearn.inspection import permutation_importance

r_sklearn = permutation_importance(
    svm_model,
    bike_test[bike_features],
    bike_test["cnt"].values,
    scoring="neg_mean_absolute_error",
    n_repeats=30,
    random_state=42
)

sklearn_pfi_df = pd.DataFrame({
    "feature": bike_features,
    "importance_mean_diff (MAE)": r_sklearn.importances_mean,
    "importance_std": r_sklearn.importances_std,
    "importance_ratio (e_perm/e_orig)": (orig_mae + r_sklearn.importances_mean) / orig_mae
}).sort_values("importance_mean_diff (MAE)", ascending=False)

print("--- 1) scikit-learn (permutation_importance) Çıktısı ---")
print(sklearn_pfi_df.to_string(index=False))
